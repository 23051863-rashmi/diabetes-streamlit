def load_notebook(nb_path):
    if not os.path.exists(nb_path):
        raise FileNotFoundError(f"Notebook not found at: {nb_path}")
    with open(nb_path, "r", encoding="utf-8") as f:
        return nbformat.read(f, as_version=4)

def extract_code_cells(nb):
    cells = []
    for cell in nb.cells:
        if cell.cell_type == "code":
            src = cell.source
            if isinstance(src, list):
                src = "".join(src)
            cells.append(src)
    return cells

def guess_imports(cells):
    imports = set()
    pattern = re.compile(r'^\s*(?:import|from)\s+([A-Za-z0-9_.]+)', re.MULTILINE)
    for c in cells:
        for m in pattern.findall(c):
            imports.add(m.split('.')[0])
    # always include streamlit
    imports.add("streamlit")
    commons = ["numpy","pandas","sklearn","joblib","torch","tensorflow","transformers","requests"]
    reqs = []
    for module in sorted(imports):
        if module in commons:
            reqs.append(module)
    for module in sorted(imports):
        if module not in commons:
            reqs.append(module)
    return reqs

def find_cells_with_keywords(cells, keywords):
    matched = []
    for i, c in enumerate(cells):
        low = c.lower()
        if any(k in low for k in keywords):
            matched.append((i, c))
    return matched
